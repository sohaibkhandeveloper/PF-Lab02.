#include<stdio.h>
int main()

{
	float radius = 7.5;
	float pi=3.14159;
	float diameter=2*radius;
	float circumference=2*pi*radius;
	float area=pi*radius*radius;
	 
	 
	 printf("==============circle geometry report=========\n");
	 printf("radius:%.3f\n",radius);
	 printf("diameter:%.3f\n",diameter);
	 printf("circumference:\t%.3f\n\t",circumference);
	 printf("area:%.3f\n",area);
	 return 0;
	
}
